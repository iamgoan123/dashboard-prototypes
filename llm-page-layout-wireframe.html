Calculated fields

-- AI share of site sessions (GA4)
-- Channel names must match your AI channel grouping exactly
SUM(CASE WHEN `AI New Channel Grouping` IN
  ("ChatGPT","Google Gemini","Perplexity",
   "Microsoft Copilot","Claude","kagi.com / referral")
  THEN Sessions ELSE 0 END) / SUM(Sessions)

-- Landing page type (GA4), adjust to the site's URL paths
CASE
  WHEN REGEXP_MATCH(`Landing page`, "^/$") THEN "Homepage"
  WHEN REGEXP_MATCH(`Landing page`, "^/products/") THEN "Product"
  WHEN REGEXP_MATCH(`Landing page`, "^/collections/") THEN "Category"
  WHEN REGEXP_MATCH(`Landing page`, "^/blogs?/") THEN "Guide"
  ELSE "Other"
END

-- Long query flag (GSC): 6 or more words
REGEXP_MATCH(Query, "^(\\S+\\s+){5,}\\S+$")

-- Question-style flag (GSC)
REGEXP_MATCH(LOWER(Query),
  "^(how|what|which|best|why|can|does|is|are|should)\\b")

-- Long-query share of impressions (GSC)
SUM(CASE WHEN REGEXP_MATCH(Query, "^(\\S+\\s+){5,}\\S+$")
  THEN Impressions ELSE 0 END) / SUM(Impressions)
